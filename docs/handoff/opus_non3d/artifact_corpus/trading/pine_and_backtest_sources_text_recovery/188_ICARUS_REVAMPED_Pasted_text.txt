//   OR via TradingView → Strategy → Properties tab.
//   NOTE: slippage is left at 0 so v1 ↔ v2 ↔ v3 backtests compare like-for-like.
//   For a live-realistic read, set Properties → Slippage = 1 tick on ALL.
//   ★ Properties → Backtesting → "Use bar magnifier" should be ON: with
//     resting brackets on 1–5m bars the emulator otherwise GUESSES whether
//     TP1 or SL printed first inside a bar — the #1 source of fake win rate.

// ══════════════════════════════════════════════════════════════════
// v3 — EVIDENCE PASS (see ICARUS_v3_RESEARCH_DOSSIER.md for sources)
//
//   Every v3 change traces to a large-sample study or an NQ-specific
//   dataset; every one is a SOFT modifier with a weight and a toggle
//   (the one hard rule — event blackout — fires a handful of days/yr).
//   Nothing in the RATE/gate/engine math changed.
//
//   RUNNER PROTECTION (§12.7) — replaces v2's immediate break-even.
//     Five independent tests (Market Stalker, Trading Rush, Quantfish,
//     setup4alpha 105k trades, r/algotrading ES) agree: arming BE at
//     ~0.33R — which is exactly v2's TP1/SL geometry — converts winners
//     into scratches and cuts expectancy ~20%, while Davey's 567k-test
//     "BE ≈ best exit" used far larger arm thresholds. v3 default:
//       1) after TP1 the runner stop moves to NET break-even — the level
//          where TP1's banked profit exactly covers the runner's loss, so
//          the TRADE can no longer lose while the runner keeps ~0.8 ATR
//          of room (entry − q1·TP1/q2 ≈ −10 pts on 5-lot defaults);
//       2) true BE+offset arms only after ≥N bars since TP1 AND price
//          has extended ≥m·ATR beyond TP1 (continuation proven);
//       3) if the regime is WEAK at TP1 (regime score < 0.40, LTF health
//          weak, or volume-delta exhaustion against the position), BE
//          arms immediately — the one case the evidence says BE helps.
//     "Immediate BE (v2)" remains selectable for A/B.
//
//   SESSION STRUCTURE BIAS (§9.16) — nqstats 10-yr NQ (2,571 days):
//     Initial Balance (09:30–10:30) closing above its own midpoint → IB
//     high breaks by the close 82.3%; below → IB low breaks 76.5%.
//     tradingstats (2,827 days): RTH open above the overnight midpoint →
//     overnight high breaks first 76.2%. Gao et al. (JFE 2018): the first
//     half-hour's return predicts the last half-hour's. Combined into
//     one directional day-prior applied as ±0.5·avg_weight.
//
//   HOUR-BREACH TIMING (§9.16) — nqstats (17,701 events): a breach of
//     the prior hour's range in minutes 0–20 reverts to the hour open
//     67–87% of the time; in minutes 40–60 only 10–33%. Applied as
//     −/+0.5·avg_weight when the entry bar is that breach.
//
//   MIDDAY (§11) — practitioner consensus + nqstats' lowest MFE/MAE
//     ratios 11:00–13:00 + Quantpedia's flat 11–12 drift: 11:30–13:00
//     ET is a gate PENALTY by default (hard block optional).
//
//   EVENT BLACKOUT (§11) — Lucca & Moench / CME / LSEG: FOMC-day
//     13:45–15:00 no new entries (editable date list); optional
//     CPI/NFP-day 09:30–10:00 block.
//
//   SIZING (§12) — Harvey et al. (vol targeting), r/algotrading ES/NQ
//     (vol-tercile sizing, −40% flat-regime DD): "Risk-$ per trade" and
//     "Vol-regime tercile" modes added. Default stays Fixed so your
//     baseline comparison is untouched.
//
//   NOT added, deliberately: ORB overlay (≈50% on index products in a
//   1.17M-trade dataset), ML/meta-labeling (matched random on NQ), more
//   indicator filters ("the filter pile is why nothing survives").
// ══════════════════════════════════════════════════════════════════

// ══════════════════════════════════════════════════════════════════
// v2 — EXECUTION-LAYER REBUILD (R.A.T.E. signal math is UNTOUCHED)
//
//   The RATE engine, the 12-vote gate, every regime/SMC/Kalman/PE/
//   structure engine and TIDE's range logic are byte-for-byte the v1
//   math. What changed is everything between "signal" and "fill":
//
//   ENTRY TIMING
//     • Arming window: a RATE flip now ARMS the direction for N bars
//       (default 3) instead of requiring the confluence gate to clear
//       on the exact flip bar. v1 silently discarded every flip where
//       the gate confirmed one bar late. A max-chase guard (×ATR from
//       the flip close) stops the window from buying extended prints.
//     • Entry window (default 09:30–15:30 ET) separate from the RTH
//       gate: no new risk opened into the close.
//     • Optional pullback-limit entry (OFF): rests a limit N×ATR below/
//       above the signal close for a few bars, with market fallback.
//
//   EXITS
//     • Bracket orders are placed on the SAME bar as the entry, in tick
//       distances relative to the actual fill (profit=/loss=). TP1/TP2/
//       SL are live from the fill tick and fill at their exact levels
//       intrabar. v1 only placed each exit AFTER a bar had already
//       crossed the level, so every fill landed one bar late at the
//       next open — stops slipped, TPs got phantom better-than-limit
//       fills, and a gap through the stop had no resting order at all.
//     • Break-even runner: after TP1 fills, the remaining position's
//       stop moves to entry + offset (default +1 pt, covers commission)
//       — converts "TP1 then full-stop" from a net loser into a net
//       winner. Optional ATR/Kalman trail ratchets ABOVE break-even.
//     • Opposite-flip exit / reversal: a RATE flip against an open RATE
//       position closes it (thesis invalidated) and, if the gate clears,
//       reverses in a single order.
//     • EOD flat (default 15:55 ET, bar-size agnostic): an intraday
//       system no longer carries 45-pt stops through the Globex reopen.
//     • Exit distances are computed on the signal bar and FROZEN — v1
//       recomputed Structure-mode targets every bar from a moving swing.
//
//   STATE INTEGRITY
//     • All position state (RATE qty/side/avg price, TIDE qty/side) is
//       derived from strategy.opentrades per entry id, every bar. v1
//       hand-tracked these and assumed exits had filled on the bar they
//       were detected; it also used strategy.position_avg_price, which
//       is the AGGREGATE average when RATE and TIDE co-hold.
//     • Vote-weight learning and time-of-day stats now read only RATE
//       closed trades (v1 booked ANY strategy.closedtrades increment —
//       including TIDE's — against the open RATE slot).
//     • A RATE reversal explicitly closes an opposing TIDE position; a
//       TIDE signal is refused on a bar where RATE is entering the other
//       way (v1 could net the two into a garbage position).
//
//   DATA CONSISTENCY
//     • HTF/LTF request.security() now uses the confirmed-bar idiom
//       (expr[1] + lookahead_on) so historical and realtime bars see the
//       same values. v1's lookahead_off returned the developing HTF bar
//       in realtime but the previous confirmed bar in the backtest.
//     • ta.stdev hoisted out of the Kalman conditional (consistent calc).
// ══════════════════════════════════════════════════════════════════

// ══════════════════════════════════════════════════════════════════
// §0  TIMEFRAME GUARD — removed.
//     The HTF/LTF system (§3) now uses independently user-configurable
//     timeframes rather than a chart-relative ladder, so the script no
//     longer assumes or requires a specific base chart timeframe.
//     Usable on any timeframe (1m through 1h+) — RTH session gating in
//     §9/§11 still applies to entries regardless of chart timeframe.
// ══════════════════════════════════════════════════════════════════

// ══════════════════════════════════════════════════════════════════
// §1.5  REAL OHLC — foundation for the toggle above
//
//     request.security() re-requesting the SAME symbol/timeframe pulls
//     the underlying market data, not the chart's HA-transformed view —
//     this is the one place a script CAN get real OHLC regardless of
//     what the Properties panel has the chart set to display.
//
//     Pine's ta.atr()/ta.dmi()/ta.supertrend() cannot accept an
//     alternate source — they always read the chart's own displayed
//     series. RATE's regime score and flip-band both depend on ATR/DMI,
//     so real independence requires hand-rebuilding Wilder's ATR and
//     DMI/ADX from real_h/real_l/real_c using ta.rma (which DOES take
//     an explicit source) rather than the ta.atr/ta.dmi built-ins.
//     Computed unconditionally (cheap); only CONSUMED when the toggle
//     above is on.
// ══════════════════════════════════════════════════════════════════
f_real_ohlc() =>
    [open, high, low, close]

[_ro_raw, _rh_raw, _rl_raw, _rc_raw] = request.security(syminfo.tickerid, timeframe.period, f_real_ohlc(), lookahead=barmerge.lookahead_off)

// FIX (audit): na-guard the fetch. If request.security ever returns na
// (data gap, symbol issue), falling back to the chart's own values is
// vastly safer than propagating na through ATR, DMI, the SuperTrend
// flip, and every downstream engine.
float real_o = nz(_ro_raw, open)
float real_h = nz(_rh_raw, high)
float real_l = nz(_rl_raw, low)
float real_c = nz(_rc_raw, close)

float real_tr = math.max(real_h - real_l, math.max(math.abs(real_h - real_c[1]), math.abs(real_l - real_c[1])))

float real_up       = real_h - real_h[1]
float real_dn       = real_l[1] - real_l
float real_plus_dm  = (real_up > real_dn and real_up > 0.0) ? real_up : 0.0
float real_minus_dm = (real_dn > real_up and real_dn > 0.0) ? real_dn : 0.0

// Master dispatch — every downstream section below reads from these
// four instead of bare open/high/low/close, so the toggle propagates
// uniformly rather than needing to be checked in dozens of places.
float _o = use_real_ohlc ? real_o : open
float _h = use_real_ohlc ? real_h : high
float _l = use_real_ohlc ? real_l : low
float _c = use_real_ohlc ? real_c : close

// ── Cross-system position state — DERIVED every bar from the broker
//    emulator's own open-trade list, routed by entry id. Declared this
//    early because TIDE's conflict check in §9.15 reads it hundreds of
//    lines before §12. v1 hand-tracked these counters and assumed each
//    exit had filled on the bar it was detected, which let them drift
//    from the real position; reading strategy.opentrades makes the
//    state exact by construction and immune to RATE/TIDE cross-talk.
//    (On the bar an entry is PLACED the position doesn't exist yet —
//    it fills at the next open — so these read 0 on that bar. Every
//    consumer below is written with that in mind.)
int   rate_qty_open  = 0      // RATE Mythos's OWN open quantity
bool  rate_pos_long  = false
float rate_avg_price = na     // RATE's OWN fill price (NOT strategy.position_avg_price — that is the aggregate incl. TIDE)
int   rate_entry_bar = na
int   h4_qty_open    = 0      // TIDE's OWN open quantity
bool  h4_pos_long    = false
float h4_avg_price   = na
if strategy.opentrades > 0
    for _oi = 0 to strategy.opentrades - 1
        string _eid = strategy.opentrades.entry_id(_oi)
        float  _sz  = strategy.opentrades.size(_oi)
        if _eid == "Long" or _eid == "Short"
            rate_qty_open  += int(math.round(math.abs(_sz)))
            rate_pos_long  := _sz > 0.0
            rate_avg_price := strategy.opentrades.entry_price(_oi)
            rate_entry_bar := strategy.opentrades.entry_bar_index(_oi)
        else if _eid == "TideLong" or _eid == "TideShort"
            h4_qty_open  += int(math.round(math.abs(_sz)))
            h4_pos_long  := _sz > 0.0
            h4_avg_price := strategy.opentrades.entry_price(_oi)

// ══════════════════════════════════════════════════════════════════
// §2  HTF / LTF HELPERS
// ══════════════════════════════════════════════════════════════════
f_st_dir() =>
    [_v, _d] = ta.supertrend(3.0, 14)
    _d

// Lightweight regime-strength proxy for cross-timeframe reads — mirrors
// the ADX component already used as the heaviest-weighted piece of the
// real rate_regime formula (§4), without recomputing that entire
// multi-engine chain 7 additional times per bar via request.security.
f_regime_proxy() =>
    [_dip, _dim, _adx] = ta.dmi(14, 14)
    math.min(math.max((_adx - 15.0) / 20.0, 0.0), 1.0)

// One combined call per HTF connection: direction + regime strength.
//
// v2 — CONFIRMED-BAR IDIOM. Every value is offset by one bar INSIDE the
// requested timeframe and fetched with lookahead_on. That combination
// returns the last fully-CLOSED bar of that timeframe on every chart
// bar, identically on historical and realtime data. The v1 form
// (no offset, lookahead_off) returned the previous confirmed HTF bar on
// historical chart bars but the still-DEVELOPING HTF bar in realtime —
// so the live HUD/gate saw fresher (and mutable) HTF state than the
// backtest ever did. There is no future leak in either form; this one
// just makes backtest == live.
f_htf_combined() =>
    [_v, _d] = ta.supertrend(3.0, 14)
    float _reg = f_regime_proxy()
    [_d[1], _reg[1]]

// One combined call per LTF connection: direction + bars since that
// timeframe's own last direction flip + regime strength. Bars-since-flip
// is the "has this LTF signal proven itself yet" read — a small value
// means the LTF direction is brand new and unproven.
f_ltf_combined() =>
    [_v, _d] = ta.supertrend(3.0, 14)
    int _bsf = ta.barssince(_d != _d[1])
    float _reg = f_regime_proxy()
    [_d[1], _bsf[1], _reg[1]]

// ══════════════════════════════════════════════════════════════════
// §3  HTF BIAS (5 independently-configurable connections)
//     + LTF SIGNAL-FAILURE DETECTION (fixed 2m / 5m)
//
//     Replaces the old chart-timeframe-relative 2-rung ladder, which
//     only worked correctly when the chart itself was locked to 1m/3m/5m
//     (its ternary silently fell through to the wrong rungs on any other
//     timeframe). Every connection here is independent of chart
//     timeframe by construction — the script now works correctly on
//     any base timeframe.
// ══════════════════════════════════════════════════════════════════
// lookahead_on is SAFE here only because f_htf_combined() returns [1]-
// offset (already-closed) values — see the note above the function.
[htf1_dir, htf1_regime] = request.security(syminfo.tickerid, htf_tf_1, f_htf_combined(), lookahead=barmerge.lookahead_on)
[htf2_dir, htf2_regime] = request.security(syminfo.tickerid, htf_tf_2, f_htf_combined(), lookahead=barmerge.lookahead_on)
[htf3_dir, htf3_regime] = request.security(syminfo.tickerid, htf_tf_3, f_htf_combined(), lookahead=barmerge.lookahead_on)
[htf4_dir, htf4_regime] = request.security(syminfo.tickerid, htf_tf_4, f_htf_combined(), lookahead=barmerge.lookahead_on)
[htf5_dir, htf5_regime] = request.security(syminfo.tickerid, htf_tf_5, f_htf_combined(), lookahead=barmerge.lookahead_on)

int htf_bull_count5 = (htf1_dir == -1 ? 1 : 0) + (htf2_dir == -1 ? 1 : 0) + (htf3_dir == -1 ? 1 : 0) + (htf4_dir == -1 ? 1 : 0) + (htf5_dir == -1 ? 1 : 0)
int htf_bear_count5 = (htf1_dir ==  1 ? 1 : 0) + (htf2_dir ==  1 ? 1 : 0) + (htf3_dir ==  1 ? 1 : 0) + (htf4_dir ==  1 ? 1 : 0) + (htf5_dir ==  1 ? 1 : 0)
// FIX (audit): nz-guarded — users can point these 5 connections at any
// timeframe, including ones with sparse/missing data. 0.5 = neutral.
float htf_regime_avg = (nz(htf1_regime, 0.5) + nz(htf2_regime, 0.5) + nz(htf3_regime, 0.5) + nz(htf4_regime, 0.5) + nz(htf5_regime, 0.5)) / 5.0

bool mtf_long  = htf_bull_count5 >= htf_min_bull_for_bias
bool mtf_short = htf_bear_count5 >= htf_min_bull_for_bias

// FIX: §6's zone-scoring formula (f_zone_score) was calibrated and
// validated assuming a 0-2 alignment range from the old 2-rung ladder —
// float(htf_a)/2.0*25.0 contributes up to 25 of the zone's 0-85 score.
// Feeding it a raw 0-5 count would double that contribution and silently
// shift every zone's E/S/W tier threshold relative to the validated
// backtest. Rescaled proportionally onto the same 0-2 range so §6's
// calibration is completely unaffected by having more HTF connections.
int htf_bull = int(math.round(float(htf_bull_count5) / 5.0 * 2.0))
int htf_bear = int(math.round(float(htf_bear_count5) / 5.0 * 2.0))

[ltf_2m_dir, ltf_2m_bsf, ltf_2m_regime] = request.security(syminfo.tickerid, "2", f_ltf_combined(), lookahead=barmerge.lookahead_on)
[ltf_5m_dir, ltf_5m_bsf, ltf_5m_regime] = request.security(syminfo.tickerid, "5", f_ltf_combined(), lookahead=barmerge.lookahead_on)

// "Failed / unproven lower-timeframe signal" — that timeframe changed
// direction within its own last N bars AND currently opposes the trade
// direction being evaluated. Informational by default; use_ltf_veto
// (off by default) opts into using this as a hard entry block.
bool ltf_fail_long  = use_ltf_check and ((ltf_2m_dir == 1  and ltf_2m_bsf <= ltf_instability_bars) or (ltf_5m_dir == 1  and ltf_5m_bsf <= ltf_instability_bars))
bool ltf_fail_short = use_ltf_check and ((ltf_2m_dir == -1 and ltf_2m_bsf <= ltf_instability_bars) or (ltf_5m_dir == -1 and ltf_5m_bsf <= ltf_instability_bars))

// ══════════════════════════════════════════════════════════════════
// §4  RATE ENGINE  +  FDI (Fractal Dimension Index)
// ══════════════════════════════════════════════════════════════════

// ── Component scores ──
[_dmi_p, _dmi_m, _dmi_adx] = ta.dmi(rate_adx_len, rate_adx_smooth)
// Hand-rolled real-OHLC DMI/ADX (Wilder's formula) — only meaningfully
// different from the native ta.dmi() call above when use_real_ohlc is on.
float real_tr_rma_adx   = ta.rma(real_tr, rate_adx_len)
float real_plus_dm_rma  = ta.rma(real_plus_dm, rate_adx_len)
float real_minus_dm_rma = ta.rma(real_minus_dm, rate_adx_len)
float real_plus_di      = real_tr_rma_adx != 0.0 ? 100.0 * real_plus_dm_rma / real_tr_rma_adx : 0.0
float real_minus_di     = real_tr_rma_adx != 0.0 ? 100.0 * real_minus_dm_rma / real_tr_rma_adx : 0.0
float real_dx           = (real_plus_di + real_minus_di) != 0.0 ? 100.0 * math.abs(real_plus_di - real_minus_di) / (real_plus_di + real_minus_di) : 0.0
float real_adx          = ta.rma(real_dx, rate_adx_smooth)

float rate_di_plus  = use_real_ohlc ? real_plus_di  : _dmi_p
float rate_di_minus = use_real_ohlc ? real_minus_di : _dmi_m
float rate_adx       = use_real_ohlc ? real_adx      : _dmi_adx
float rate_adx_score = math.min(math.max((rate_adx - 15.0) / 20.0, 0.0), 1.0)

// ── ATR fast/slow — dispatch between native ta.atr() and real-OHLC RMA(TR) ──
float _rate_atr_s_real = ta.rma(real_tr, rate_atr_fast)
float _rate_atr_s_chart = ta.atr(rate_atr_fast)
float rate_atr_s    = use_real_ohlc ? _rate_atr_s_real : _rate_atr_s_chart
float _rate_atr_l_real = ta.rma(real_tr, rate_atr_slow)
float _rate_atr_l_chart = ta.atr(rate_atr_slow)
float rate_atr_l    = use_real_ohlc ? _rate_atr_l_real : _rate_atr_l_chart
float rate_vol_ratio = rate_atr_l > 0.0 ? rate_atr_s / rate_atr_l : 1.0
float rate_vol_score = math.min(math.max((rate_vol_ratio - 0.5) / 1.0, 0.0), 1.0)

float rate_net = math.abs(_c - _c[rate_eff_len])
float rate_sum = 0.0
for _ri = 0 to rate_eff_len - 1
    rate_sum += math.abs(_c[_ri] - _c[_ri + 1])
float rate_eff       = rate_sum > 0.0 ? rate_net / rate_sum : 0.0
float rate_eff_score = math.min(math.max((rate_eff - 0.1) / 0.35, 0.0), 1.0)

int   rate_dir  = _c > _c[1] ? 1 : -1
float rate_cons = 0.0
for _ri = 0 to rate_eff_len - 1
    rate_cons += ((_c[_ri] > _c[_ri + 1]) == (rate_dir > 0)) ? 1.0 : 0.0
float rate_cons_score = rate_cons / float(rate_eff_len)

// ── FDI: independent loop (works even when fdi_len ≠ rate_eff_len) ──
// Formula: FDI = 1 + log(path/net) / log(2*N)
// FDI ≈ 1.0 = clean trend | FDI ≈ 1.5 = random/noisy
float fdi_path = 0.0
for _fi = 0 to fdi_len - 1
    fdi_path += math.abs(_c[_fi] - _c[_fi + 1])
float fdi_net = math.max(math.abs(_c - _c[fdi_len]), 0.0001)
float fdi_raw = fdi_path > 0.0 ? 1.0 + math.log(fdi_path / fdi_net) / math.log(2.0 * fdi_len) : 1.5
float fdi     = ta.ema(fdi_raw, fdi_smooth)
fdi           := math.max(1.0, math.min(2.0, fdi))

// ── FDI adjusts regime component weights (weights always sum to 1.0) ──
// Low FDI (trending): trust ADX + efficiency more. High FDI (noisy): trust volatility more.
float fdi_t    = math.max(0.0, math.min(1.0, (fdi - 1.0) / 0.5))  // 0=trending → 1=noisy
float w_adx    = 0.30 + (1.0 - fdi_t) * 0.10   // 0.40 → 0.30
float w_vol    = 0.30 +        fdi_t  * 0.10   // 0.30 → 0.40
float w_eff    = 0.25 + (1.0 - fdi_t) * 0.05   // 0.30 → 0.25
float w_cons   = 0.15 -        fdi_t  * 0.05   // 0.15 → 0.10

float rate_regime = ta.ema(
    rate_adx_score  * w_adx  +
    rate_vol_score  * w_vol  +
    rate_eff_score  * w_eff  +
    rate_cons_score * w_cons, 5)

// ── FDI widens ATR mult in noisy markets, tightens in trending ──
float fdi_mult_adj  = 1.0 + (fdi - 1.0) * 0.30
float rate_atr_mult = (rate_max_mult - (rate_max_mult - rate_min_mult) * rate_regime) * fdi_mult_adj
rate_atr_mult       := math.max(rate_min_mult, math.min(rate_max_mult, rate_atr_mult))

// ── SuperTrend ──

float _rate_atr_val_chart = ta.atr(rate_atr_len)
float rate_atr_val   = use_real_ohlc ? _rate_atr_val_real : _rate_atr_val_chart

float rate_lower_raw = rate_hl2 - rate_atr_mult * rate_atr_val

    rate_p_lower := rate_lower_raw
    rate_st_line := rate_lower_raw

    else

else

// ── Hurst Exponent — multi-window log-log regression (audit #4) ──
// The previous implementation used a SINGLE window R/S ratio mapped
// through log(R/S)/log(n). That is a crude, noisy, biased estimator:
// true Hurst is the SLOPE of log(R/S) against log(n) across multiple
// window sizes, not one point evaluated at a single n. Estimated here
// across 4 windows by least-squares slope — the textbook rescaled-
// range method, materially more stable.
//
// Mean and standard deviation are computed MANUALLY inside the loop
// rather than via ta.sma/ta.stdev: those require a simple-int length,
// which a user-function parameter does not reliably satisfy.
f_rs_ratio(int len) =>
    float _sum = 0.0
    for _i = 0 to len - 1
        _sum += (_c[_i] - _c[_i + 1])
    float _mean = _sum / float(len)
    float _cum = 0.0
    float _cmax = 0.0
    float _cmin = 0.0
    float _sq = 0.0
    for _i = 0 to len - 1
        float _dev = (_c[_i] - _c[_i + 1]) - _mean
        _cum  := _cum + _dev
        _cmax := math.max(_cmax, _cum)
        _cmin := math.min(_cmin, _cum)
        _sq   += _dev * _dev
    float _R = _cmax - _cmin
    float _S = math.sqrt(_sq / float(len))
    (_R > 0.0 and _S > 0.0) ? _R / _S : 1.0

float _rs1 = f_rs_ratio(10)
float _rs2 = f_rs_ratio(20)
float _rs3 = f_rs_ratio(30)
float _rs4 = f_rs_ratio(40)

float _lx1 = math.log(10.0)
float _lx2 = math.log(20.0)
float _lx3 = math.log(30.0)
float _lx4 = math.log(40.0)
float _ly1 = math.log(math.max(_rs1, 0.0001))
float _ly2 = math.log(math.max(_rs2, 0.0001))
float _ly3 = math.log(math.max(_rs3, 0.0001))
float _ly4 = math.log(math.max(_rs4, 0.0001))

float _lxm = (_lx1 + _lx2 + _lx3 + _lx4) / 4.0
float _lym = (_ly1 + _ly2 + _ly3 + _ly4) / 4.0
float _hnum = (_lx1-_lxm)*(_ly1-_lym) + (_lx2-_lxm)*(_ly2-_lym) + (_lx3-_lxm)*(_ly3-_lym) + (_lx4-_lxm)*(_ly4-_lym)
float _hden = (_lx1-_lxm)*(_lx1-_lxm) + (_lx2-_lxm)*(_lx2-_lxm) + (_lx3-_lxm)*(_lx3-_lxm) + (_lx4-_lxm)*(_lx4-_lxm)
float hurst_raw = _hden > 0.0 ? _hnum / _hden : 0.5
hurst_raw := math.max(0.0, math.min(1.0, hurst_raw))

float hurst = ta.ema(hurst_raw, 5)

    rate_regime_str := "STRONG TREND"
else if rate_regime >= 0.65 and hurst <= 0.55
    rate_regime_str := "WEAK TREND"
else if rate_regime >= 0.40 and hurst > 0.55
    rate_regime_str := "EMERGING"
else if rate_regime >= 0.40 and hurst < 0.40
    rate_regime_str := "CHOPPY"
else if hurst < 0.40
    rate_regime_str := "MEAN-REV"

bool regime_accel = (rate_regime - rate_regime[3]) > 0.02

// §5  EHLERS HOMODYNE DISCRIMINATOR — Dominant Cycle Phase

    phase_angle := phase_raw
else if I1 < 0.0 and Q1 >= 0.0
    phase_angle := math.pi - phase_raw
else if I1 < 0.0 and Q1 < 0.0
    phase_angle := math.pi + phase_raw
else
    phase_angle := 2.0 * math.pi - phase_raw
